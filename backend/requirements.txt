# Core conversion library with specific format extras.
# Using named extras instead of [all] to avoid heavy deps (ONNX, Whisper, Azure)
# that cause OOM on Render's 512MB free tier.
#
# Extras included:
#   [pdf]   → pdfminer.six, pdfplumber (PDF text extraction)
#   [docx]  → mammoth (Word documents)
#   [pptx]  → python-pptx (PowerPoint slides)
#   [xlsx]  → openpyxl (Excel .xlsx — THIS is what registers XlsxConverter)
#   [xls]   → xlrd (legacy Excel .xls files)
#
# Without [xlsx], MarkItDown raises MissingDependencyException even if
# openpyxl is installed separately — the extra is what registers the converter.
markitdown[pdf,docx,pptx,xlsx,xls]>=0.1.0

# HTML conversion
beautifulsoup4
markdownify

# API framework
fastapi>=0.109.0
uvicorn[standard]>=0.27.0
python-multipart>=0.0.9

# Image processing (metadata fallback for images)
Pillow>=10.0.0
